# Next.js Powerful Middleware Note

# 1. What is Middleware ?

Middleware is a function that runs:

```txt id="m1x7qa"
BEFORE request reaches page or API
```

Middleware can:

* Check Login
* Check Role
* Redirect User
* Block User
* Rate Limit
* Modify Request
* Protect Routes

---

# 2. Middleware Full Flow

```txt id="z9f2ra"
Browser Request
        │
        ▼
┌─────────────────────┐
│    middleware.js    │
└─────────────────────┘
        │
        ├── Check Auth
        ├── Check Role
        ├── Check Token
        ├── Check Rate Limit
```

# 25. Complete Auth Middleware

## middleware.js

```js id="m8k2zd"
import { NextResponse } from "next/server";

export function middleware(req) {

    const token =
        req.cookies.get("token")?.value;

    const path =
        req.nextUrl.pathname;

    // PRIVATE ROUTES
    const privateRoutes = [
        "/dashboard",
        "/profile",
    ];

    const isPrivate =
        privateRoutes.includes(path);

    // NOT LOGIN
    if (!token && isPrivate) {

        return NextResponse.redirect(
            new URL("/login", req.url)
        );
    }

    return NextResponse.next();
}
```

# 28. Complete Public + Private Middleware

```js id="x7k3tz"
import { NextResponse } from "next/server";

export function middleware(req) {

    const token =
        req.cookies.get("token")?.value;

    const path =
        req.nextUrl.pathname;

    const publicRoutes = [
        "/login",
        "/register",
    ];

    const isPublic =
        publicRoutes.includes(path);

    // USER NOT LOGIN
    if (!token && !isPublic) {

        return NextResponse.redirect(
            new URL("/login", req.url)
        );
    }

    // USER ALREADY LOGIN
    if (token && isPublic) {

        return NextResponse.redirect(
            new URL("/dashboard", req.url)
        );
    }

    return NextResponse.next();
}
```

# 31. Role Based Middleware

## middleware.js

```js id="t1q9cn"
import { NextResponse } from "next/server";

export function middleware(req) {

    const role =
        req.cookies.get("role")?.value;

    const path =
        req.nextUrl.pathname;

    // ADMIN PAGE
    if (
        path.startsWith("/admin")
        &&
        role !== "admin"
    ) {

        return NextResponse.redirect(
            new URL("/403", req.url)
        );
    }

    return NextResponse.next();
}
```

# 35. Simple Rate Limit Middleware

## middleware.js

```js id="p4x8sw"
import { NextResponse } from "next/server";

// MEMORY STORE
const ipRequests = new Map();

export function middleware(req) {

    const ip =
        req.ip || "unknown";

    const now = Date.now();

    const windowTime = 60 * 1000;

    const maxRequest = 5;

    // GET OLD DATA
    const requestData =
        ipRequests.get(ip);

    if (!requestData) {

        ipRequests.set(ip, {
            count: 1,
            startTime: now,
        });

        return NextResponse.next();
    }

    // RESET WINDOW
    if (
        now - requestData.startTime
        >
        windowTime
    ) {

        ipRequests.set(ip, {
            count: 1,
            startTime: now,
        });

        return NextResponse.next();
    }

    // INCREMENT COUNT
    requestData.count++;

    // BLOCK USER
    if (
        requestData.count
        >
        maxRequest
    ) {

        return NextResponse.json(
            {
                message:
                    "Too Many Requests",
            },
            {
                status: 429,
            }
        );
    }

    return NextResponse.next();
}
```
