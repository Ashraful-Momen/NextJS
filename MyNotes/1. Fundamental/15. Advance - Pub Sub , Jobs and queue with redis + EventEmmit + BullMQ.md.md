
Below is a **simple but production-style Next.js architecture** that combines:

* Prisma (DB)
* Service Layer (business logic)
* EventEmitter (observer system)
* Redis Queue (background jobs)
* Worker (async processing)
* Email + Logging system

No TSX, clean JS.

# 6. Service Layer (Main Business Logic)

## services/userService.js

```js id="a6"
import { prisma } from "../lib/prisma.js";
import { eventBus } from "../lib/eventBus.js";

export async function createUser(data) {

    // 1. DB operation
    const user = await prisma.user.create({
        data
    });

    console.log("User created in DB");

    // 2. Emit observer event
    eventBus.emit("user.created", user);

    return user;
}
```

# 8. Observer Listener (Event System)

## observers/userObserver.js

```js id="a8"
import { eventBus } from "../lib/eventBus.js";
import { emailQueue } from "../lib/queue.js";

// OBSERVER: USER CREATED
eventBus.on("user.created", async (user) => {

    console.log("Observer triggered for user:", user.id);

    // push job to queue (DO NOT block request)
    await emailQueue.add("sendWelcomeEmail", {
        email: user.email,
        name: user.name,
    });
});
```

---

# 9. Queue Worker (Background Processing)

## worker/emailWorker.js

```js id="a9"
import { Worker } from "bullmq";
import IORedis from "ioredis";

const connection = new IORedis();

const worker = new Worker(
    "emailQueue",
    async (job) => {

        if (job.name === "sendWelcomeEmail") {

            const { email, name } = job.data;

            console.log("Sending email to:", email);

            // simulate email send
            await sendEmail(email, name);
        }
    },
    { connection }
);

async function sendEmail(email, name) {

    console.log(`Email sent to ${name} <${email}>`);
}
```

# 10. Logger System (Observer Side Effect)

## lib/logger.js

```js id="a10"
export function logEvent(message, data) {

    console.log("[LOG]", message, data);
}
```

---

## Attach Logger to Observer

```js id="a11"
import { eventBus } from "../lib/eventBus.js";
import { logEvent } from "../lib/logger.js";

eventBus.on("user.created", (user) => {

    logEvent("USER_CREATED", user);
});
```
