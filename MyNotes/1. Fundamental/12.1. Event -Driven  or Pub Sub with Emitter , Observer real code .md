# 5. Observer Equivalent #1 → Prisma Middleware

## middleware/prismaObserver.js

```js id="p2"
import { prisma } from "../lib/prisma.js";

prisma.$use(async (params, next) => {

    // USER CREATE OBSERVER
    if (
        params.model === "User" &&
        params.action === "create"
    ) {
        console.log("Observer: User is being created");
    }

    // USER UPDATE OBSERVER
    if (
        params.model === "User" &&
        params.action === "update"
    ) {
        console.log("Observer: User is updating");
    }

    return next(params);
});

export default prisma;
```

# 6. Observer Equivalent #2 → Service Layer (Recommended)

## services/userService.js

```js id="s1"
import { prisma } from "../lib/prisma.js";

export async function createUser(data) {

    const user = await prisma.user.create({
        data
    });

    // Observer-like side effects
    console.log("Service: User Created");

    await sendWelcomeEmail(user.email);
    await logUserActivity(user.id);

    return user;
}
```

---

## Helper functions


---

## Emit Event (Service Layer)

```js id="e2"
import { eventBus } from "../lib/eventBus.js";

export async function createUser(data) {

    const user = await prisma.user.create({ data });

    // emit event
    eventBus.emit("user.created", user);

    return user;
}
```

---

## Listen Event (Observer)

```js id="e3"
import { eventBus } from "../lib/eventBus.js";

eventBus.on("user.created", (user) => {

    console.log("EVENT: User Created", user.id);

    sendEmail(user.email);
});
```

# 8. Observer Equivalent #4 → Server Action

## actions/userAction.js

```js id="a1"
"use server";

import { prisma } from "../lib/prisma.js";

export async function createUserAction(formData) {

    const name = formData.get("name");
    const email = formData.get("email");

    const user = await prisma.user.create({
        data: { name, email }
    });

    // side effects (observer-like)
    console.log("Server Action: User Created");

    return user;
}
```

# 11. Key Difference Summary

| Layer             | Role                    |
| ----------------- | ----------------------- |
| Prisma Middleware | DB-level observer       |
| Service Layer     | Business logic observer |
| EventEmitter      | Decoupled observer      |
| Server Action     | UI trigger observer     |

---

# 12. Best Practice (Important)

```txt id="b1"
✔ Use Service Layer as main logic
✔ Use EventEmitter for async tasks
✔ Use Prisma middleware only for logging/audit
❌ Avoid putting business logic inside middleware
```

# 13. Final Simple Definition

```txt id="b2"
Next.js Observer = Combination of:
Service Layer + Events + Middleware + Server Actions
```
