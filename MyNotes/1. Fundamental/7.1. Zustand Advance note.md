# 50. Auth Store Example

## store/authStore.js

```js id="k2m9ra"
import { create } from "zustand";

export const useAuthStore =
    create((set) => ({

        user: null,

        token: null,

        login: (userData, token) =>

            set({

                user: userData,

                token: token,
            }),

        logout: () =>

            set({

                user: null,

                token: null,
            }),
    }));
```

---

# 51. Cart Store Example

## store/cartStore.js

```js id="n8v3qw"
import { create } from "zustand";

export const useCartStore =
    create((set) => ({

        cartItems: [],

        addToCart: (product) =>

            set((state) => ({

                cartItems: [

                    ...state.cartItems,

                    product,
                ],
            })),

        clearCart: () =>

            set({
                cartItems: [],
            }),
    }));
```

# 52. Theme Store Example

# 57. Why Zustand Simpler Than Redux ?

Redux needs:

```txt id="m9v2tx"
store
reducers
actions
dispatch
slice
provider
```

Zustand needs:

```txt id="j4n8qw"
create()
set()
```

ONLY.

# 60. Zustand Component Example

## components/Navbar.jsx

```jsx id="v5n9qb"
"use client";

import {
    useAuthStore
}
from "@/store/authStore";

import {
    useCartStore
}
from "@/store/cartStore";

export default function Navbar() {

    const user =
        useAuthStore(
            (state) => state.user
        );

    const cartItems =
        useCartStore(
            (state) =>
                state.cartItems
        );

    return (

        <div>

            <h1>
                {user?.name}
            </h1>

            <h2>

                Cart:
                {cartItems.length}

            </h2>

        </div>
    );
}
```

# 66. Example

## authStore.js

```js id="c5m9vw"
import { create } from "zustand";

import {
    useCartStore
}
from "./cartStore";

export const useAuthStore =
    create((set) => ({

        logout: () => {

            // CLEAR CART ALSO
            useCartStore
                .getState()
                .clearCart();

            set({

                user: null,
            });
        },
    }));
```

# 72. Combine Slice Store

## store/store.js

```js id="q4v7ra"
import { create } from "zustand";

import {
    createAuthSlice
}
from "./slices/authSlice";

import {
    createCartSlice
}
from "./slices/cartSlice";

export const useAppStore =

    create((...a) => ({

        ...createAuthSlice(...a),

        ...createCartSlice(...a),
    }));
```
